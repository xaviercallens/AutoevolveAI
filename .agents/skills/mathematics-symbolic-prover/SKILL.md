## 3. Reference Implementation: Discrete Hodge Exterior Calculus

```python
"""Discrete Exterior Calculus (DEC) Discrete Hodge Laplacian with d^2 = 0 nilpotency."""
from __future__ import annotations
import numpy as np
import scipy.sparse as sp

def build_simplicial_complex_2d(n_pts: int = 50) -> tuple[sp.csr_matrix, sp.csr_matrix]:
    """Generates discrete exterior derivative operators d0 (0-forms -> 1-forms) and d1 (1-forms -> 2-forms)."""
    # d0: Vertices -> Edges (incidence matrix)
    # d1: Edges -> Faces (curl/exterior derivative)
    # Exact topological invariant: d1 @ d0 == 0
    from scipy.spatial import Delaunay
    points = np.random.RandomState(42).uniform(-1.0, 1.0, size=(n_pts, 2))
    tri = Delaunay(points)
    
    # Extract unique edges
    edges_set = set()
    for simplex in tri.simplices:
        for i in range(3):
            e = tuple(sorted([simplex[i], simplex[(i + 1) % 3]]))
            edges_set.add(e)
    edges = sorted(list(edges_set))
    edge_to_idx = {e: idx for idx, e in enumerate(edges)}
    
    # Construct d0: n_edges x n_vertices
    row, col, data = [], [], []
    for idx, (u, v) in enumerate(edges):
        row.extend([idx, idx])
        col.extend([u, v])
        data.extend([-1.0, 1.0])
    d0 = sp.csr_matrix((data, (row, col)), shape=(len(edges), n_pts), dtype=np.float64)
    
    # Construct d1: n_faces x n_edges
    row1, col1, data1 = [], [], []
    for f_idx, simplex in enumerate(tri.simplices):
        # Directed boundary
        for i in range(3):
            u, v = simplex[i], simplex[(i + 1) % 3]
            e = tuple(sorted([u, v]))
            e_idx = edge_to_idx[e]
            sign = 1.0 if u < v else -1.0
            row1.append(f_idx)
            col1.append(e_idx)
            data1.append(sign)
    d1 = sp.csr_matrix((data1, (row1, col1)), shape=(len(tri.simplices), len(edges)), dtype=np.float64)
    
    # Assert nilpotency invariant d1 @ d0 = 0
    d_squared = d1.dot(d0)
    nilpotency_error = float(np.max(np.abs(d_squared.data))) if d_squared.nnz > 0 else 0.0
    assert nilpotency_error < 1e-14, f"Coboundary nilpotency violated: ||d^2|| = {nilpotency_error}"
    return d0, d1
```
