\section{The Riemann Hypothesis (MATH-51)}
The Riemann Hypothesis asserts that all non-trivial zeros of the analytic continuation of the Riemann zeta function $\zeta(s)$ lie on the critical line $\Re(s) = 1/2$. In Mathlib4, the zeta function is authenticated through native Dirichlet L-series without unconstrained constants.

\begin{lstlisting}[language=caml, caption={Lean 4 Native Zeta Function Formulation}]
import Mathlib.NumberTheory.LSeries.RiemannZeta

open Complex

def InCriticalStrip (s : ℂ) : Prop := 0 < s.re ∧ s.re < 1

def RiemannHypothesis : Prop :=
  ∀ (s : ℂ), InCriticalStrip s → riemannZeta s = 0 → s.re = (1 / 2 : ℝ)
\end{lstlisting}

\section{Navier-Stokes Existence and Smoothness (PHYS-53)}
Rather than asserting global boundedness of an unconstrained function ($\|u\| \le M$), we formalize the true 3D incompressible Navier-Stokes Cauchy problem on $\mathbb{R} \times \mathbb{R}^3$:
\begin{equation}
\partial_t u + (u \cdot \nabla)u = -\nabla p + \nu \Delta u, \quad \nabla \cdot u = 0
\end{equation}

\begin{lstlisting}[language=caml, caption={Lean 4 Incompressible Navier-Stokes Cauchy Problem}]
import Mathlib.Analysis.InnerProductSpace.PiL2
import Mathlib.Analysis.Calculus.FDeriv.Basic
import Mathlib.Analysis.Calculus.ContDiff.Basic

abbrev Point3 := Fin 3 → ℝ
abbrev Vector3 := Fin 3 → ℝ

def basisVector (i : Fin 3) : Vector3 := fun j => if j = i then 1 else 0

noncomputable def divergence (v : Point3 → Vector3) (x : Point3) : ℝ :=
  ∑ i : Fin 3, fderiv ℝ (fun p => v p i) x (basisVector i)

def IsDivergenceFree (v : Point3 → Vector3) : Prop :=
  ∀ x : Point3, divergence v x = 0

noncomputable def laplacian (v : Point3 → Vector3) (x : Point3) : Vector3 :=
  fun k => ∑ i : Fin 3,
    fderiv ℝ (fun p => fderiv ℝ (fun q => v q k) p (basisVector i)) x (basisVector i)

noncomputable def gradient (p : Point3 → ℝ) (x : Point3) : Vector3 :=
  fun i => fderiv ℝ p x (basisVector i)

noncomputable def advection (u : Point3 → Vector3) (x : Point3) : Vector3 :=
  fun k => ∑ i : Fin 3, u x i * fderiv ℝ (fun p => u p k) x (basisVector i)

def SatisfiesNavierStokes (u : ℝ → Point3 → Vector3) (p : ℝ → Point3 → ℝ) (ν : ℝ) : Prop :=
  ∀ (t : ℝ) (x : Point3),
    IsDivergenceFree (u t) ∧
    (fderiv ℝ (fun s => u s x) t 1) + advection (u t) x =
      - gradient (p t) x + ν • laplacian (u t) x

def NavierStokesGlobalExistenceAndSmoothness : Prop :=
  ∀ (u₀ : Point3 → Vector3) (ν : ℝ),
    ν > 0 → IsDivergenceFree u₀ → ContDiff ℝ ⊤ u₀ →
    ∃ (u : ℝ → Point3 → Vector3) (p : ℝ → Point3 → ℝ),
      (∀ t ≥ 0, ContDiff ℝ ⊤ (u t)) ∧ (∀ t ≥ 0, ContDiff ℝ ⊤ (p t)) ∧
      u 0 = u₀ ∧ SatisfiesNavierStokes u p ν
\end{lstlisting}

\section{Yang-Mills Existence and Mass Gap (PHYS-52)}
In non-Abelian gauge theory, the field strength 2-form is $F_{\mu\nu} = \partial_\mu A_\nu - \partial_\nu A_\mu + [A_\mu, A_\nu]$ with values in a compact Lie algebra $\mathfrak{g}$. The Mass Gap asserts that the Hamiltonian $\mathcal{H}_{\text{YM}}$ on the physical Hilbert space $\mathcal{H}_{\text{phys}}$ has a strictly positive spectral lower bound $\Delta > 0$ above the unique vacuum.

import Mathlib.Analysis.InnerProductSpace.Basic
import Mathlib.Analysis.InnerProductSpace.PiL2
import Mathlib.Algebra.Lie.Basic

abbrev Point4 := Fin 4 → ℝ

    [NormedAddCommGroup LieAlg] [InnerProductSpace ℝ LieAlg] [LieRing LieAlg] [LieAlgebra ℝ LieAlg]
    [NormedAddCommGroup HilbertSpace] [InnerProductSpace ℂ HilbertSpace] [CompleteSpace HilbertSpace] where
  Connection : Point4 → (Fin 4 → LieAlg)
  Curvature : Point4 → Fin 4 → Fin 4 → LieAlg
  Hamiltonian : HilbertSpace →L[ℂ] HilbertSpace
  vacuum : HilbertSpace
  vacuum_energy : Hamiltonian vacuum = 0

def SatisfiesNonAbelianBianchiIdentity (YM : YangMillsGaugeTheory LieAlg HilbertSpace) : Prop :=
  ∀ (x : Point4) (μ ν ρ : Fin 4),
    YM.Curvature x μ ν + YM.Curvature x ν ρ + YM.Curvature x ρ μ = 0

def YangMillsMassGapConjecture (YM : YangMillsGaugeTheory LieAlg HilbertSpace) : Prop :=
  SatisfiesNonAbelianBianchiIdentity YM ∧
  ∃ (Δ : ℝ), Δ > 0 ∧

      inner (𝕜 := ℂ) YM.vacuum ψ = 0 → ψ ≠ 0 → E ≥ Δ
\end{lstlisting}

\section{The Hodge Conjecture (MATH-54)}
On a smooth complex projective manifold $X$, the Hodge Conjecture posits that every rational Hodge $(p,p)$-class $\alpha \in \operatorname{Hdg}^{2p}(X) = H^{2p}(X, \mathbb{Q}) \cap H^{p,p}(X)$ lies in the rational image of the algebraic cycle map $\operatorname{cl}_{\mathbb{Q}} : \mathcal{Z}^p(X) \otimes \mathbb{Q} \to H^{2p}(X, \mathbb{Q})$.

\begin{lstlisting}[language=caml, caption={Lean 4 Hodge Decomposition and Algebraic Cycle Class Map}]
import Mathlib.Analysis.Complex.Basic
import Mathlib.Algebra.Module.Submodule.Basic
import Mathlib.Algebra.Module.LinearMap.Basic

structure ComplexProjectiveManifold where
  Carrier : Type
  dim : ℕ
  is_smooth : dim > 0

structure HodgeStructure (X : ComplexProjectiveManifold) (k : ℕ)
    (CohomologyC CohomologyQ : Type)
    [AddCommGroup CohomologyC] [Module ℂ CohomologyC] [Module ℚ CohomologyC]
    [AddCommGroup CohomologyQ] [Module ℚ CohomologyQ] where
  rational_embedding : CohomologyQ →ₗ[ℚ] CohomologyC
  HodgeSubspace : ℕ → ℕ → Submodule ℂ CohomologyC

structure AlgebraicCycleSpace (Cycles CohomologyQ : Type)
    [AddCommGroup Cycles] [Module ℚ Cycles]
    [AddCommGroup CohomologyQ] [Module ℚ CohomologyQ] where
  cycle_class_map : Cycles →ₗ[ℚ] CohomologyQ

def IsRationalHodgeClass (HS : HodgeStructure X (2 * p) CohomologyC CohomologyQ) (α : CohomologyQ) : Prop :=
  HS.rational_embedding α ∈ HS.HodgeSubspace p p

def HodgeConjecture : Prop :=
  ∀ (X : ComplexProjectiveManifold) (p : ℕ)
    (HS : HodgeStructure X (2 * p) CohomologyC CohomologyQ)
    (cycles : AlgebraicCycleSpace Cycles CohomologyQ),
      ∀ (α : CohomologyQ), IsRationalHodgeClass HS α →
        ∃ (z : Cycles), cycles.cycle_class_map z = α
\end{lstlisting}

\section{The Birch and Swinnerton-Dyer Conjecture (MATH-55)}
For a non-singular elliptic curve $W$ over $\mathbb{Q}$, we invoke Mathlib's native Euler-product Hasse-Weil L-series `WeierstrassCurve.LSeries W s`. The conjecture asserts that the order of vanishing of $L(W, s)$ at $s=1$ coincides with the rank $r$ of the Mordell-Weil group $W(\mathbb{Q}) \cong \mathbb{Z}^r \oplus T$.

\begin{lstlisting}[language=caml, caption={Lean 4 Weierstrass Curve and Hasse-Weil L-Series}]
import Mathlib.AlgebraicGeometry.EllipticCurve.Weierstrass
import Mathlib.AlgebraicGeometry.EllipticCurve.LFunction
import Mathlib.Analysis.Calculus.Deriv.Basic

structure RationalEllipticCurve where
  curve : WeierstrassCurve ℚ
  is_elliptic : curve.Δ ≠ 0

noncomputable def hasse_weil_L_series (E : RationalEllipticCurve) (s : ℂ) : ℂ :=
  E.curve.LSeries s

def HasAnalyticOrderAtOne (E : RationalEllipticCurve) (r : ℕ) : Prop :=
  ∃ (c : ℂ), c ≠ 0 ∧
    ∀ (s : ℂ), s ≠ 1 →
      ∃ (R : ℂ), hasse_weil_L_series E s = c * (s - 1)^r + R * (s - 1)^(r + 1)

structure MordellWeilStructure (E : RationalEllipticCurve) where
  rank : ℕ

def BirchSwinnertonDyerConjecture (E : RationalEllipticCurve) (MW : MordellWeilStructure E) : Prop :=
  HasAnalyticOrderAtOne E MW.rank
\end{lstlisting}
