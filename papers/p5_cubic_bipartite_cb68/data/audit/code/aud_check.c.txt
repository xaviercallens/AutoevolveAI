/* aud_check.c -- auditor's own streaming certificate checker for EG-CB68 certificates.
 * usage: aud_check EXPECTED_CAP EXPECTED_ORBIT < stream
 * The header must be exactly "R <cap> <orbit>" (no flags); C6 records are refused; completions are refused.
 * Schedule re-derived by the auditor's own candidate generator (same as aud_search.c).
 * Prints counters and an FNV-1a hash of the decision transcript (same encoding as producer). */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>
typedef uint64_t u64;
#define NP 64
static int CAP, ORB;
static int nblocks, npoints, bl[NP][3], inc[NP][3], deg[NP];
static u64 c_st, c_att, c_str, c_8, c_16;
static u64 H = 1469598103934665603ULL;
static long long ln = 0;
static char buf[8192];
static void mix(u64 x) { for (int i = 0; i < 8; i++) { H ^= (x >> (8 * i)) & 0xff; H *= 1099511628211ULL; } }
static void fail(const char *m) { printf("REJECT line %lld: %s\n", ln, m); exit(1); }
static int next(void) { if (!fgets(buf, sizeof buf, stdin)) return 0; ln++; return 1; }
static int collinear(int x, int y) {
    for (int i = 0; i < deg[x]; i++) { int b = inc[x][i]; for (int j = 0; j < 3; j++) if (bl[b][j] == y) return 1; }
    return 0;
}
static void put(int a, int b, int c) {
    if (deg[a] > 2 || deg[b] > 2 || deg[c] > 2 || !(a < b && b < c) || c >= NP) fail("bad insertion");
    int k = nblocks++; bl[k][0] = a; bl[k][1] = b; bl[k][2] = c;
    inc[a][deg[a]++] = k; inc[b][deg[b]++] = k; inc[c][deg[c]++] = k;
    if (c + 1 > npoints) npoints = c + 1;
}
static void take(int a, int b, int c, int o) { nblocks--; deg[a]--; deg[b]--; deg[c]--; npoints = o; }
static int edge(int u, int v) { /* u point, v block id (>=64) or vice versa */
    if (u >= 64) { int t = u; u = v; v = t; }
    if (u < 0 || u >= npoints || v < 64 || v - 64 >= nblocks) return 0;
    int b = v - 64; return bl[b][0] == u || bl[b][1] == u || bl[b][2] == u;
}
typedef struct { int q, r; } cand;
static int cmpc(const void *x, const void *y) { const cand *u = x, *v = y; return u->q != v->q ? u->q - v->q : u->r - v->r; }
static void state(void) {
    c_st++;
    int p = 0; while (p < npoints && deg[p] == 3) p++;
    if (p == npoints) fail("complete state (counterexample candidate) in stream");
    int lq = -1, lr = -1;
    for (int i = 0; i < deg[p]; i++) { int k = inc[p][i]; if (bl[k][0] != p) continue;
        if (bl[k][1] > lq || (bl[k][1] == lq && bl[k][2] > lr)) { lq = bl[k][1]; lr = bl[k][2]; } }
    int oldn = npoints, pool[NP + 2], np = 0;
    for (int x = p + 1; x < oldn; x++) if (deg[x] < 3 && !collinear(p, x)) pool[np++] = x;
    int nold = np; pool[np++] = oldn; pool[np++] = oldn + 1;
    cand cs[2400]; int nc = 0;
    for (int i = 0; i < np; i++) for (int j = i + 1; j < np; j++) {
        int q = pool[i], r = pool[j];
        if (i >= nold && !(q == oldn && r == oldn + 1)) continue;
        if (j >= nold && i < nold && r != oldn) continue;
        if (r >= CAP) continue;
        if (!(q > lq || (q == lq && r > lr))) continue;
        cs[nc].q = q; cs[nc].r = r; nc++;
    }
    qsort(cs, nc, sizeof(cand), cmpc);
    for (int t = 0; t < nc; t++) {
        int q = cs[t].q, r = cs[t].r;
        if (!next()) fail("truncated");
        c_att++;
        char tag[4]; int a, b, off = 0;
        if (sscanf(buf, "%3s %d %d%n", tag, &a, &b, &off) != 3 || tag[1]) fail("malformed record");
        if (a != q || b != r) fail("schedule mismatch");
        u64 code = ((u64)q << 16) | ((u64)r << 8);
        if (tag[0] == 'S') {
            if (buf[off] != '\n') fail("junk after S");
            if (!(r < oldn && collinear(q, r))) fail("false structural");
            c_str++; mix(code | 1);
        } else if (tag[0] == '8' || tag[0] == 'G') {
            int L = tag[0] == '8' ? 6 : 14, path[40], n = 0, used, v; char *s = buf + off;
            while (n < 40 && sscanf(s, "%d%n", &v, &used) == 1) { path[n++] = v; s += used; }
            while (*s == ' ') s++;
            if (*s != '\n' || n != L + 1) fail("witness length/format");
            int m[3] = {p, q, r};
            int e0 = path[0], e1 = path[L], ok0 = 0, ok1 = 0;
            for (int i = 0; i < 3; i++) { if (m[i] == e0) ok0 = 1; if (m[i] == e1) ok1 = 1; }
            if (!ok0 || !ok1 || e0 == e1 || e0 >= oldn || e1 >= oldn) fail("witness endpoints");
            for (int i = 0; i < n; i++) for (int j = i + 1; j < n; j++) if (path[i] == path[j]) fail("witness not simple");
            for (int i = 0; i < L; i++) {
                int u = path[i], w = path[i + 1];
                if ((i % 2 == 0) != (u < 64)) fail("witness parity");
                if (!edge(u, w)) fail("witness edge absent");
            }
            if (tag[0] == '8') { c_8++; mix(code | 2); } else { c_16++; mix(code | 3); }
        } else if (tag[0] == 'E') {
            if (buf[off] != '\n') fail("junk after E");
            mix(code | 4);
            put(p, q, r); state(); take(p, q, r, oldn);
            if (!next() || strcmp(buf, "U\n")) fail("missing U");
        } else fail("unknown/forbidden tag");
    }
}
int main(int argc, char **argv) {
    if (argc != 3) return 2;
    CAP = atoi(argv[1]); ORB = atoi(argv[2]);
    char want[64]; snprintf(want, sizeof want, "R %d %d\n", CAP, ORB);
    if (!next() || strcmp(buf, want)) fail("header differs from expected parameters");
    if (ORB == 0) put(0, 1, 2);
    else if (ORB == 1 || ORB == 2) { put(0, 1, 3); put(1, 2, 4); put(0, 2, 5); if (ORB == 1) put(0, 4, 6); else put(0, 6, 7); }
    else return 2;
    state();
    if (!next()) fail("missing END");
    unsigned long long e[7];
    if (sscanf(buf, "END %llu %llu %llu %llu %llu %llu %llu", e, e + 1, e + 2, e + 3, e + 4, e + 5, e + 6) != 7) fail("END format");
    if (e[0] != c_st || e[1] != c_att || e[2] != c_str || e[3] != 0 || e[4] != c_8 || e[5] != c_16 || e[6] != 0) fail("END counters");
    if (next()) fail("trailing data");
    printf("ACCEPT cap=%d orbit=%d states=%llu attempted=%llu structural=%llu c8=%llu c16=%llu completions=0 hash=%016llx\n",
           CAP, ORB, (unsigned long long)c_st, (unsigned long long)c_att, (unsigned long long)c_str,
           (unsigned long long)c_8, (unsigned long long)c_16, (unsigned long long)H);
    return 0;
}
