/* comp2g6.c -- completion lines "v a,b,c ..." -> Levi graph6 (points 0..v-1, blocks v..2v-1), only lines with
 * v == WANT (argv[1]); verifies 3-regularity and linearity; prints graph6 to stdout. */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
int main(int argc, char **argv) {
    int want = atoi(argv[1]);
    static char line[1 << 16];
    long bad = 0;
    while (fgets(line, sizeof line, stdin)) {
        char *s = line; int v = (int)strtol(s, &s, 10);
        if (v != want) continue;
        int n = 2 * v, deg[64] = {0}; static unsigned char A[128][128]; memset(A, 0, sizeof A);
        static unsigned char pr[64][64]; memset(pr, 0, sizeof pr);
        for (int k = 0; k < v; k++) {
            int a = (int)strtol(s, &s, 10); s++; int b = (int)strtol(s, &s, 10); s++; int c = (int)strtol(s, &s, 10);
            int t[3] = {a, b, c};
            for (int i = 0; i < 3; i++) { deg[t[i]]++; A[t[i]][v + k] = A[v + k][t[i]] = 1;
                for (int j = i + 1; j < 3; j++) { if (pr[t[i]][t[j]]) bad++; pr[t[i]][t[j]] = pr[t[j]][t[i]] = 1; } }
        }
        for (int i = 0; i < v; i++) if (deg[i] != 3) bad++;
        char out[2048]; int o = 0; out[o++] = (char)(63 + n);
        int bits = 0, val = 0;
        for (int j = 1; j < n; j++) for (int i = 0; i < j; i++) {
            val = (val << 1) | A[i][j]; bits++;
            if (bits == 6) { out[o++] = (char)(63 + val); bits = 0; val = 0; }
        }
        if (bits) { val <<= (6 - bits); out[o++] = (char)(63 + val); }
        out[o++] = '\n'; fwrite(out, 1, o, stdout);
    }
    if (bad) { fprintf(stderr, "INVALID completions: %ld\n", bad); return 1; }
    return 0;
}
