def jpowLoop : Nat → Nat → J → J → J

  | f + 1, k, x, v =>
    if k = 0 then v else

def absI (a : IV) : IV := if 0 ≤ a.u then a else if a.v ≤ 0 then neg a else ⟨0, max (-a.u) a.v⟩

/-- `IV::mid`: `[⌊(u+v)/2⌋, ⌈(u+v)/2⌉]`. -/
def midI (a : IV) : IV := ⟨(a.u + a.v) / 2, -((-(a.u + a.v)) / 2)⟩

def sumI : List IV → IV := List.foldl add z0
def sumJ : List J → J := List.foldl jadd jz

/-- `cpol(t, coeff)` (core.hpp:173): re-expansion of the polynomial at the midpoint, Horner in `h = t - mid`. -/
def cpol (t : J) (coeff : List IV) : J :=
  let m := midI t.a0
  let h := jaddi t (neg m)
  let n := coeff.length
  let cc := (List.range n).map fun j =>
    sumI ((List.range (n - j)).map fun i =>
      mul (mul (rat (comb (j + i) j : Int) 1) (coeff.getD (j + i) z0)) (powI m i))
  cc.foldr (fun c z => jaddi (jmul z h) c) (jc z0)
